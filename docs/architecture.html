核心数据结构

字段只保留跨生命周期或跨模块需要的状态；交接结果直接复用步骤输出。

// 专家：描述能力，不携带密钥或执行代码
type ExpertManifest = {
  apiVersion: "expert/v1"
  kind: "expert"
  metadata: {
    id: string              // 稳定引用与版本入口
    version: string         // 发布版本
    author: string          // 归属与审核主体
    tags: string[]          // 市场检索
  }
  spec: {
    instructions: string   // 专家职责与边界
    capabilities: string[] // 可发现能力
  }
}

// 专家团：成员声明角色，步骤声明依赖
  type TeamManifest = {
   metadata: ExpertMetadata
   spec: {
     members: { id, expert, role }[] // 候选专家与角色
     steps: { member, dependsOn? }[] // 安全依赖边界，不代表必调
   }
 }

 // 统筹 Agent 每轮看到的上下文
 type TeamCoordinationInput = {
   task: string
   candidates: TeamCandidate[]
   completed: Record
   dispatched: memberId[]
   skipped: memberId[]
   availableMembers: memberId[]
 }

 // 统筹 Agent 每轮的结构化决定
 type TeamDispatchPlan = {
   calls: { member, task }[]
   skipMembers: memberId[]
   rationale: string
 }

 // 运行时交接：真实结果与跳过缺口分开
 type ExpertExecutionInput = {
   task: string
   assignment: string
   member: TeamMember
   expert: ExpertManifest
   completed: Record<memberId, ExpertResult>
   skipped: memberId[]
 }

边界与失败模式

把“能不能发布”和“能不能执行”分成两次判断，避免市场成为不受控的代码执行入口。

01
上传者提交声明默认只接受 manifest；缺少 ID、版本、职责或能力时拒绝。
02
统筹 Agent 选择本轮统筹 Agent 查看候选职责、任务和已完成结果，决定调用哪些专家、分配什么子任务以及跳过哪些成员。
03
Runner 校验调度决定运行器拒绝越权成员、重复决定和未收口依赖；同一轮调用可以并行，下一轮等待结果后再规划。
04
宿主注入执行能力模型、工具、密钥、网络和文件权限由宿主执行器决定；跳过成员没有结果，缺口必须显式传递。
05
最终结果取声明末步最后一个步骤是最终集成步骤，不能跳过；剩余候选在最终结果生成后不再调用。
